% ========== SLIDE 2: PROBLEM =========================================
\begin{frame}{The problem}
  \noindent\small\textbf{What the system does.} A user asks a research question; the tool
  collects evidence from Wikipedia, arXiv and web search at the same time and prints an LLM
  answer whose numbered citations point to that evidence. A failing source becomes a warning,
  not a crash.

  \vspace{0.2em}
  \begin{center}
  \begin{tikzpicture}[x=1cm,y=1cm,
      src/.style={stage, draw=teal, fill=teal!10, minimum width=2.9cm, minimum height=0.6cm},
      answer/.style={stage, draw=ok, fill=ok!10}]
    \node[stage] (q) at (0,0) {\faQuestionCircle\ Question\\[-1pt]{\scriptsize CLI, up to 2000 chars}};
    \node[src] (w) at (4.3,0.75) {\faWikipediaW\ Wikipedia};
    \node[src] (a) at (4.3,0)    {\faIcon{file-alt}\ arXiv};
    \node[src] (s) at (4.3,-0.75) {\faSearch\ Web search};
    \node[stage] (l) at (8.3,0) {\faBrain\ LLM synthesis\\[-1pt]{\scriptsize Gemini}};
    \node[answer] (r) at (11.7,0) {\faQuoteRight\ Answer\\[-1pt]{\scriptsize with [1], [2], \dots}};
    \foreach \n in {w,a,s}{\draw[flow] (q.east) -- (\n.west); \draw[flow] (\n.east) -- (l.west);}
    \draw[flow] (l.east) -- (r.west);
    \node[font=\scriptsize\itshape, text=teal] at (4.3,1.32) {concurrent, one deadline per source};
  \end{tikzpicture}
  \end{center}

  \vspace{0.1em}
  {\footnotesize
  \textbf{Inputs:} the question (up to 2000 characters), \texttt{-{}-sources} from \texttt{wiki},
  \texttt{arxiv}, \texttt{web}, optional \texttt{-{}-no-cache}; keys and limits from \texttt{.env}.\\[3pt]
  \textbf{Outputs:} answer, numbered references and per-source warnings; exit code 0 (answer),
  1 (none) or 2 (invalid input).\\[3pt]
  \textbf{Entry point:} CLI only, no HTTP server:\enspace
  {\scriptsize\texttt{python -m researcher ask "..." -{}-sources wiki,arxiv,web}}}
\end{frame}

% ========== SLIDE 5: KEY DESIGN DECISIONS ============================
\begin{frame}{Key design decisions}
  \begin{tikzpicture}[x=1cm,y=1cm]
    \decision{-0.9}{1}{asyncio over threads, one deadline per source}
      {All the work is waiting on at most three sources, so one event loop is enough.}
      {Rejected: threads, and \texttt{TaskGroup}, which cancels the other sources when one fails.}
    \decision{-3.0}{2}{Protocols and constructor injection around the unchanged \texttt{ai/}}
      {Tests swap in fakes without patching, and mypy still checks the signatures.}
      {Trade-off accepted: one wiring function (\texttt{run\_ask}) must build everything in order.}
    \decision{-5.1}{3}{SQLite cache of source evidence, not of answers}
      {Every run still synthesizes; repeated questions skip the network.}
      {We would change it for many parallel users (single-file writer) or if answers became expensive.}
  \end{tikzpicture}
\end{frame}

% ========== SLIDE 6: CONCURRENCY (THE BENCHMARK) =====================
\begin{frame}{Concurrency: sequential vs.\ concurrent}
  \noindent{\small\textbf{Workload:} the 5 supplied questions $\times$ all 3 sources, cache off.
  Offline: simulated delays 0.1\,/\,0.2\,/\,0.3\,s, median of 5 runs.
  Live: real Wikipedia, arXiv and Tavily requests, one pair per question.\par}

  \vspace{0.3em}
  \begin{columns}[T,onlytextwidth]
  \column{0.56\textwidth}
  \footnotesize
  \begin{tabularx}{\linewidth}{@{}Xcc@{}}
    \toprule
    & \textbf{Sequential} & \textbf{Concurrent} \\
    \midrule
    Offline wall-clock (s)      & 0.623--0.628 & 0.312--0.314 \\
    Offline speedup             & 1.0$\times$  & 1.99--2.00$\times$ \\
    Live wall-clock (s)         & 4.41--7.04   & 3.62--5.60 \\
    Live speedup (q1--q5)       & 1.0$\times$  & 0.79--1.82$\times$ \\
    Provider 429s observed      & 0 & 0 \\
    Failed or timed-out sources & 0 & 0 \\
    \bottomrule
  \end{tabularx}

  \column{0.42\textwidth}
  \begin{tikzpicture}[x=1cm,y=1.2cm]
    \node[font=\scriptsize\bfseries, anchor=south] at (2.55,2.15) {Live speedup per question};
    \draw[black!40] (0,0) -- (5.1,0);
    \draw[dashed, black!50] (0,1) -- (5.1,1);
    \node[font=\tiny, anchor=east, text=black!60] at (-0.05,1) {1$\times$};
    \foreach \i/\v/\c in {0/1.66/teal, 1/0.79/amber, 2/1.80/teal, 3/1.05/teal, 4/1.82/teal}{
      \fill[\c] (0.2+\i*1.0,0) rectangle (0.8+\i*1.0,\v);
      \node[font=\tiny\bfseries, anchor=north, text=white] at (0.5+\i*1.0,\v-0.04) {\v$\times$};
      \node[font=\tiny, anchor=north] at (0.5+\i*1.0,0) {q\the\numexpr\i+1\relax};
    }
  \end{tikzpicture}
  \end{columns}

  \vspace{0.3em}
  {\footnotesize\textbf{Bottleneck:} the slowest source sets the concurrent time (0.3\,s offline);
  live, network latency and arXiv's 3\,s request spacing.\\[2pt]
  \textbf{Why not $3\times$:} with unequal delays the ideal is $2\times$
  (sum\,/\,max = 0.6\,/\,0.3); live, q2 was slower concurrently and Wikipedia found nothing for
  full questions.\\[2pt]
  \textit{Reproduce:} \texttt{uv run python benchmarks/benchmark\_sources.py -{}-repeats 5}}
\end{frame}

% ========== SLIDE 8: CACHE EVIDENCE ==================================
\begin{frame}{Caching: measured, not assumed}
  \begin{columns}[T,onlytextwidth]
  \column{0.55\textwidth}
  \textbf{Schema and policy}
  \begin{itemize}\small\setlength\itemsep{0.3em}
    \item Table \texttt{cache\_entries}: key \texttt{(source, query\_key)}, its automatic
      unique index, JSON \texttt{payload}; no foreign keys
    \item Key = source + normalised question; TTL 24\,h
    \item All rows derived; the external sources stay authoritative
    \item \texttt{-{}-no-cache} never opens the SQLite file
  \end{itemize}

  \column{0.42\textwidth}
  \textbf{Hit rate} = hits\,/\,lookups
  \vspace{0.4em}

  \begin{tikzpicture}[x=1cm,y=1cm]
    \statTile{0}{1.85}{amber}{\large}{0\,/\,15}{first run}
    \statTile{1.98}{1.85}{teal}{\large}{15\,/\,15}{second run}
    \statTile{3.96}{1.85}{accent}{\large}{50\%}{combined, 15\,/\,30}
  \end{tikzpicture}
  \vspace{0.3em}

  {\scriptsize 5 questions $\times$ 3 sources through the real CLI; providers simulated;
  an answer is synthesized on every run.}
  \end{columns}

  \vspace{0.6em}
  \textbf{Staleness:} empty results are also kept for 24\,h, and the key ignores provider settings.

  \vspace{0.3em}
  {\footnotesize\textit{Reproduce:} \texttt{uv run python tests/cache\_verification.py}
  \enspace\textit{(docs/cache-verification.md)}}
\end{frame}

% ========== SLIDE 11: LIMITATIONS + FUTURE ===========================
\begin{frame}{Limitations \& next steps}
  \begin{columns}[T,onlytextwidth]
  \column{0.48\textwidth}
  {\large\bfseries\color{amber}Where the system is brittle}
  \begin{itemize}\setlength\itemsep{0.5em}
    \item Full questions often find no Wikipedia match; there is no query rewriting.
    \item No fallback LLM: if Gemini is down, we show evidence but no answer.
    \item The cache keeps empty results for 24\,h, ignores provider settings, and never
      evicts rows.
  \end{itemize}

  \column{0.48\textwidth}
  {\large\bfseries\color{ok}With another week}
  \begin{enumerate}\setlength\itemsep{0.5em}
    \item Turn the question into a short topic query before searching Wikipedia.
    \item Add a settings fingerprint to the cache key, a shorter TTL for empty results, and a
      purge command.
    \item Repeat the live benchmark at least three times and compare distributions.
  \end{enumerate}
  \end{columns}
\end{frame}
